```bash
release_dir=$(mktemp -d /tmp/smart-list-release.XXXXXX)
DEVELOPER_DIR=/Applications/Xcode-beta.app/Contents/Developer \
  xcodebuild \
  -project ios/SmartShoppingList/SmartShoppingList.xcodeproj \
  -scheme SmartShoppingList \
  -configuration Release \
  -destination 'generic/platform=iOS' \
  -derivedDataPath "$release_dir/DerivedData" \
  -resultBundlePath "$release_dir/Release.xcresult" \
  build
```
